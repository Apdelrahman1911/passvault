def checked_path(p):
    p = Path(p)
    if not p.is_absolute() or ".." in p.parts: raise ValueError("invalid absolute path")
    for q in list(p.parents)[::-1] + [p]:
        if stat.S_ISLNK(q.lstat().st_mode): raise ValueError("symlink: " + str(q))
    return p

def identity(p):
    checked_path(p)
    fd = os.open(str(p), os.O_RDONLY | os.O_NOFOLLOW)
    try:
        before = os.fstat(fd)
        if not stat.S_ISREG(before.st_mode): raise ValueError("not regular: " + str(p))
        h = hashlib.sha256()
        while True:
            b = os.read(fd, 1024 * 1024)
            if not b: break
            h.update(b)
        after = os.fstat(fd)
        fields = ("st_dev", "st_ino", "st_size", "st_mtime_ns", "st_ctime_ns", "st_mode")
        if any(getattr(before, k) != getattr(after, k) for k in fields):
            raise ValueError("changed during read: " + str(p))
        end = p.lstat()
        if any(getattr(after, k) != getattr(end, k) for k in fields):
            raise ValueError("path changed: " + str(p))
        return {"bytes": after.st_size, "sha256": h.hexdigest(),
                "execute_bits": stat.S_IMODE(after.st_mode) & 0o111}
    finally:
        os.close(fd)

def read_json(p):
    first = identity(p)
    data = p.read_bytes()
    if hashlib.sha256(data).hexdigest() != first["sha256"] or identity(p) != first:
        raise ValueError("metadata changed: " + str(p))
    return json.loads(data)

def member(root, rel):
    if not isinstance(rel, str) or not rel or Path(rel).is_absolute() or ".." in Path(rel).parts:
        raise ValueError("invalid member")
    return root / rel

def run(label, argv, cwd=D, timeout=60, retain_text=True):

    record["commands"].append(item)
    p = subprocess.Popen(argv, cwd=str(cwd), env=ENV, stdout=subprocess.PIPE,
                         stderr=subprocess.PIPE, start_new_session=True)
    item["pid"] = p.pid
    item["pgid_at_birth"] = p.pid
    try:
        out, err = p.communicate(timeout=timeout)
    except BaseException:
        if p.poll() is None:
            os.killpg(p.pid, signal.SIGTERM)
            record["stopped_processes"].append({"pid": p.pid, "reason": "own live child interrupted"})
        try:
            out, err = p.communicate(timeout=5)
        except subprocess.TimeoutExpired:
            if p.poll() is None:
                os.killpg(p.pid, signal.SIGKILL)
            out, err = p.communicate(timeout=5)
        item["interrupted"] = True
        item["exit_code"] = p.returncode
        raise
    item["exit_code"] = p.returncode
    item["finished"] = datetime.datetime.now(datetime.timezone.utc).isoformat()
    for key, data in (("stdout", out), ("stderr", err)):
        item[key + "_sha256"] = hashlib.sha256(data).hexdigest()
        item[key + "_bytes"] = len(data)
        if retain_text: item[key] = data.decode("utf-8", "replace")
    if p.returncode != 0: raise ValueError(label + " exit " + str(p.returncode))
    return out
