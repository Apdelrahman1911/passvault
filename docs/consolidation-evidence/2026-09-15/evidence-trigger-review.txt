==mobile-store-release.yml==
name: Mobile Store Release

on:
  workflow_call:
    inputs:
      platform:
        required: true
        type: string
      channel:
        required: true
        type: string
      operation:
        required: true
        type: string
      version:
        required: true
        type: string
      build_number:
        required: true
        type: string
      candidate_tag:
        required: false
        default: ''
        type: string
      candidate_commit:
        required: false
        default: ''
        type: string
      testflight_distribution:
        required: false
        default: email-list
        type: string
      production_confirmation:
        required: false
        default: ''
        type: string
      testing_confirmation:
        required: false
        default: ''
        type: string
      review_withdrawal_confirmation:
        required: false
        default: ''
        type: string
      pipeline_lock_held:
        description: The protected caller already holds the repository-wide release-pipeline lock
        required: false
        default: false
        type: boolean
  workflow_dispatch:
    inputs:
      platform:
        description: Platform to build and upload
        required: true
        default: both
        type: choice
        options:
          - both
          - android
          - ios
      channel:
        description: Store destination
        required: true
        default: internal
        type: choice
        options:
          - internal
          - external
          - production
      operation:
        description: Upload a new internal binary or promote the exact existing build
        required: true
        default: upload
        type: choice
        options:
          - upload
          - promote
          - store-assets
      version:
        description: Exact marketing version; blank uses version.properties
        required: false
        type: string
      build_number:
        description: Exact committed Store build number from version.properties
        required: true
        type: string
      candidate_tag:
        description: Required readiness-approved candidate tag for production
        required: false
        type: string
      testflight_distribution:
        description: External iOS invitation method
        required: true
        default: email-list
        type: choice
        options:
          - email-list
      production_confirmation:
        description: Type I_APPROVE_PRODUCTION only for a production upload
        required: false
        type: string
      testing_confirmation:
        description: Type I_CONFIRM_REQUIRED_TESTING_COMPLETED only after required store testing is complete
        required: false
        type: string
      review_withdrawal_confirmation:
        description: Type I_APPROVE_REVIEW_WITHDRAWAL to withdraw review and replace iOS store assets
        required: false
        type: string

permissions:
  contents: read

concurrency:
  # Direct retries share the repository-wide pipeline lock. Reusable calls get
  # a run-local child lock because their protected parent already holds it.
  group: ${{ inputs.pipeline_lock_held && format('passvault-mobile-child-{0}', github.run_id) || 'passvault-release-pipeline' }}
  cancel-in-progress: false

env:
  JAVA_VERSION: '17'
  RUBY_VERSION: '3.3'
  FASTLANE_OPT_OUT_USAGE: '1'
  GRADLE_OPTS: '-Dorg.gradle.daemon=false -Dorg.gradle.workers.max=2 -Xmx4096m'
  REQUEST_PLATFORM: ${{ inputs.platform }}
  REQUEST_CHANNEL: ${{ inputs.channel }}
  REQUEST_OPERATION: ${{ inputs.operation }}
  REQUEST_VERSION: ${{ inputs.version }}
  REQUEST_BUILD_NUMBER: ${{ inputs.build_number }}
  REQUEST_CANDIDATE_COMMIT: ${{ inputs.candidate_commit }}
  REQUEST_TESTFLIGHT_DISTRIBUTION: ${{ inputs.testflight_distribution }}
  REQUEST_PRODUCTION_CONFIRMATION: ${{ inputs.production_confirmation }}
  REQUEST_TESTING_CONFIRMATION: ${{ inputs.testing_confirmation }}
  REQUEST_REVIEW_WITHDRAWAL_CONFIRMATION: ${{ inputs.review_withdrawal_confirmation }}
