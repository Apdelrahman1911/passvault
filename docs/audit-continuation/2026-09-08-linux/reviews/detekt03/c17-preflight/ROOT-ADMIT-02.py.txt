def pin(s):
    return {**{k: getattr(s, 'st_' + k) for k in ('dev', 'ino', 'uid', 'mode', 'nlink')},
            'bytes': s.st_size, 'mtime_ns': s.st_mtime_ns, 'ctime_ns': s.st_ctime_ns}

def capture(path):
    before = os.lstat(path)
    target = path
    extra = {}
    if stat.S_ISLNK(before.st_mode):
        assert path == Path('/usr/bin/python3') and os.readlink(path) == 'python3.12'
        target = Path('/usr/bin/python3.12')
        extra = {'link_pin': pin(before), 'link_target': 'python3.12'}
    fd = os.open(target, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC)
    try:
        original = os.fstat(fd)
        assert stat.S_ISREG(original.st_mode) and original.st_uid == 0
        assert original.st_nlink == 1 and not original.st_mode & 0o022
        assert original.st_size <= 32 * 1024 ** 2
        data = bytearray()
        while part := os.read(fd, 65536):
            data.extend(part)
            assert len(data) <= 32 * 1024 ** 2
        assert len(data) == original.st_size
        assert pin(original) == pin(os.fstat(fd)) == pin(os.lstat(target))
        assert pin(before) == pin(os.lstat(path))
        return bytes(data), {'sha256': hashlib.sha256(data).hexdigest(), 'pin': pin(original), **extra}
    finally:
        os.close(fd)

def write_new(path, raw):
    fd = os.open(path, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
    try:
        view = memoryview(raw)
        while view:
            n = os.write(fd, view)
            assert n > 0
            view = view[n:]
        os.fsync(fd)
    finally:
        os.close(fd)

    return (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
