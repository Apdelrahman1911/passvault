def reject_duplicates(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise Hold("duplicate JSON key")
        result[key] = value
    return result


def decode_json(raw, cap):
    if not isinstance(raw, bytes) or not raw or len(raw) > cap:
        raise Hold("empty or over-limit JSON input")
    try:
        return json.loads(
            raw.decode("utf-8"), object_pairs_hook=reject_duplicates,
            parse_constant=lambda value: (_ for _ in ()).throw(Hold("nonfinite JSON input")),
        )
    except (UnicodeError, ValueError, TypeError, RecursionError) as error:
        raise Hold("malformed JSON input") from error


def valid_absolute_path(path):
    if not isinstance(path, str) or not path.startswith("/") or len(path.encode("utf-8")) > 2048:
        raise Hold("bounded absolute path required")
    if "\x00" in path or "\n" in path or "\r" in path or path.endswith("/"):
        raise Hold("unsafe pathname")
    if any(part in ("", ".", "..") for part in path.split("/")[1:]):
        raise Hold("noncanonical pathname component")
    return path


def stable_source_bytes(parent_fd, name, cap):
    """Single original no-follow leaf capture; caller must bind all ancestors.

    This is not a standalone workspace or filesystem-admission primitive.
    An ambiguous close remains HOLD and is not retried.
    """
    if not isinstance(name, str) or "/" in name or name in ("", ".", ".."):
        raise Hold("one admitted source leaf required")
    descriptor = os.open(name, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC, dir_fd=parent_fd)
    try:
        before = os.fstat(descriptor)
        if not stat.S_ISREG(before.st_mode) or before.st_nlink != 1 or before.st_size > cap:
            raise Hold("source is not a bounded single-link regular leaf")
        blocks = []
        remaining = cap + 1
        while remaining:
            block = os.read(descriptor, min(65536, remaining))
            if not block:
                break
            blocks.append(block)
            remaining -= len(block)
        raw = b"".join(blocks)
        after = os.fstat(descriptor)
        keys = ("st_dev", "st_ino", "st_size", "st_mtime_ns", "st_ctime_ns", "st_mode", "st_uid")
        if len(raw) > cap or any(getattr(before, key) != getattr(after, key) for key in keys):
            raise Hold("source changed during capture")
        return raw
    finally:
        try:
            os.close(descriptor)
        except OSError as error:
            raise Hold("original source descriptor close failed; no retry") from error


def require_digest(actual, expected, label):
    if not isinstance(expected, str) or SHA256.fullmatch(expected) is None:
        raise Hold("invalid expected digest: " + label)
    if hashlib.sha256(actual).hexdigest() != expected:
        raise Hold("source digest mismatch: " + label)

def validate_inventory(raw, build):
    """Exact identity/argv SET validation, independent of registration order."""
    valid_absolute_path(build)
    inventory = decode_json(raw, XML_CAP)
    if not isinstance(inventory, dict) or inventory.get("kind") != "ctestInfo":
        raise Hold("wrong CTest inventory kind")
    if inventory.get("version") != {"major": 1, "minor": 0}:
        raise Hold("wrong CTest inventory schema")
    tests = inventory.get("tests")
    if not isinstance(tests, list) or len(tests) != len(CASES):
        raise Hold("missing or unexpected inventory case")
    expected = {case.name: case.command(build) for case in CASES}
    seen = set()
    for test in tests:
        if not isinstance(test, dict):
            raise Hold("malformed inventory case")
        name = test.get("name")
        if not isinstance(name, str) or name not in expected or name in seen:
            raise Hold("unknown or duplicate inventory identity")
        command = test.get("command")
        if not isinstance(command, list) or tuple(command) != expected[name]:
            raise Hold("wrong executable/selector vector")
        properties = test.get("properties", [])
        if not isinstance(properties, list):
            raise Hold("malformed test properties")
        for prop in properties:
            if not isinstance(prop, dict) or prop.get("name") != "WORKING_DIRECTORY" or prop.get("value") != build:
                raise Hold("unadmitted CTest property/environment/timeout override")
        seen.add(name)
    if seen != set(expected):
        raise Hold("case inventory identity mismatch")
    return tuple(case.name for case in CASES)


def integer_attribute(element, name, default=None):
    value = element.attrib.get(name, default)
    if not isinstance(value, str) or DECIMAL.fullmatch(value) is None:
        raise Hold("malformed JUnit count: " + name)
    return int(value)


def validate_case_xml(raw, selected, actual_exit):
    """A prospective XML case outcome, never worker/cleanup or family closure."""
    if selected not in CASES or type(actual_exit) is not int or actual_exit != 0:
        raise Hold("selected case did not exit zero")
    if not isinstance(raw, bytes) or not raw or len(raw) > XML_CAP:
        raise Hold("empty/over-limit XML")
    try:
        text = raw.decode("utf-8")
        if "<!DOCTYPE" in text.upper() or "<!ENTITY" in text.upper() or "\x00" in text:
            raise Hold("unsafe XML declaration or encoding")
        root = ET.fromstring(text)
    except (UnicodeError, ET.ParseError, RecursionError) as error:
        raise Hold("malformed JUnit XML") from error
    if root.tag != "testsuite" or integer_attribute(root, "tests") != 1:
        raise Hold("wrong JUnit suite or case count")
    for name in ("failures", "errors", "skipped", "disabled"):
        if integer_attribute(root, name, "0") != 0:
            raise Hold("failed/error/skipped/disabled JUnit outcome")
    cases = list(root.iter("testcase"))
    if len(cases) != 1 or cases[0].attrib.get("name") != selected.name:
        raise Hold("wrong or duplicate JUnit selected-case identity")
    if cases[0].attrib.get("status", "run") != "run":
        raise Hold("case did not run")
    if any(node.tag in ("failure", "error", "skipped") for node in root.iter()):
        raise Hold("nonpassing nested JUnit outcome")
    return {"case": selected.name, "xml_sha256": hashlib.sha256(raw).hexdigest(), "case_outcome": "PASS_CANDIDATE"}


def resource_floor_check(available_disk, total_ram, available_ram, *, launching):
    """Arithmetic only; MAC-MEMORY still lacks an accepted platform observer."""
    if any(type(value) is not int or value < 0 for value in (available_disk, total_ram, available_ram)):
        raise Hold("invalid resource unit/count")
    if total_ram == 0 or available_ram > total_ram:
        raise Hold("invalid physical-memory relationship")
    disk_floor, ram_percent = (12 * GIB, 25) if launching else (8 * GIB, 20)
    if available_disk < disk_floor or available_ram * 100 < total_ram * ram_percent:
        raise Hold("handoff resource floor not met")
