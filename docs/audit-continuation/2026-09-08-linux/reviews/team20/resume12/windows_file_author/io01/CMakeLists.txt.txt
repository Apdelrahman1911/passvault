cmake_minimum_required(VERSION 3.25)

project(passvault_biometric VERSION 1.0.0 LANGUAGES C CXX)

include(CTest)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(APPLE)
    enable_language(OBJCXX)
    add_library(passvault_biometric SHARED src/macos/passvault_biometric_macos.mm)
    target_compile_options(passvault_biometric PRIVATE
        -fobjc-arc
        -fvisibility=hidden
        -fstack-protector-strong
        -Wall
        -Wextra
        -Werror
    )
    target_link_options(passvault_biometric PRIVATE
        -Wl,-dead_strip
    )
    target_link_libraries(passvault_biometric PRIVATE
        "-framework Foundation"
        "-framework LocalAuthentication"
        "-framework Security"
    )
    set_target_properties(passvault_biometric PROPERTIES
        OUTPUT_NAME "passvault_biometric"
        INSTALL_NAME_DIR "@loader_path"
        BUILD_WITH_INSTALL_NAME_DIR ON
        MACOSX_RPATH OFF
    )
    if(BUILD_TESTING)
        add_executable(passvault_biometric_macos_security_test
            tests/macos/passvault_biometric_macos_security_test.mm
        )
        target_include_directories(passvault_biometric_macos_security_test PRIVATE include)
        target_compile_definitions(passvault_biometric_macos_security_test PRIVATE
            PASSVAULT_BIOMETRIC_BUILD=1
        )
        target_compile_options(passvault_biometric_macos_security_test PRIVATE
            -fobjc-arc
            -fstack-protector-strong
            -Wall
            -Wextra
            -Werror
        )
        target_link_libraries(passvault_biometric_macos_security_test PRIVATE
            "-framework Foundation"
            "-framework LocalAuthentication"
            "-framework Security"
        )
        add_test(
            NAME passvault_biometric_macos_security
            COMMAND passvault_biometric_macos_security_test
        )
        # Caller supplies an admitted private PASSVAULT_NATIVE_TEST_PARENT.
        # Separate real-filesystem cases; no default /tmp or parent cleanup.
        foreach(fixture_case IN ITEMS normal early_return cpp_exception)
            add_test(
                NAME passvault_biometric_macos_fixture_${fixture_case}
                COMMAND passvault_biometric_macos_security_test --fixture-case ${fixture_case}
            )
        endforeach()
    endif()
elseif(WIN32)
    add_library(passvault_biometric SHARED
        src/windows/passvault_biometric_windows.cpp
        src/windows/passvault_biometric_windows.rc
    )
    target_compile_definitions(passvault_biometric PRIVATE
        NOMINMAX
        UNICODE
        _UNICODE
        WIN32_LEAN_AND_MEAN
        WINVER=0x0A00
        _WIN32_WINNT=0x0A00
    )
    target_compile_options(passvault_biometric PRIVATE
        /W4
        /WX
        /guard:cf
        /sdl
    )
    target_link_options(passvault_biometric PRIVATE
        /DYNAMICBASE
        /HIGHENTROPYVA
        /NXCOMPAT
        /guard:cf
    )
    set_property(TARGET passvault_biometric PROPERTY
        MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
    )
    target_link_libraries(passvault_biometric PRIVATE
        advapi32
        bcrypt
        ole32
        runtimeobject
        windowsapp
    )
    set_target_properties(passvault_biometric PROPERTIES OUTPUT_NAME "passvault_biometric")

    if(BUILD_TESTING)
        # Compile the reviewed Windows implementation into a test-only
        # executable as well, so its pure envelope/crypto/validation paths can
        # run on CI without requiring Windows Hello enrollment.
        add_executable(passvault_biometric_windows_security_test
            tests/windows/passvault_biometric_windows_security_test.cpp
        )
        set(windows_security_targets passvault_biometric_windows_security_test)
        option(PASSVAULT_AUDIT_PVA036_037_CONTROLS
            "Build isolated historical-writer and live PRK allocation controls" OFF)
        if(PASSVAULT_AUDIT_PVA036_037_CONTROLS)
            if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
                message(FATAL_ERROR "The audit controls require MSVC x64")
            endif()
            # This is an additional, exact historical source input, not a
            # reconstructed writer or a replacement for the production library.
            set(pva036_source "${CMAKE_CURRENT_SOURCE_DIR}/../../../docs/audit-continuation/2026-09-08-linux/reviews/native/HISTORICAL-PVA036-WINDOWS.cpp.txt")
            file(SHA256 "${pva036_source}" pva036_source_sha256)
            if(NOT pva036_source_sha256 STREQUAL
                "4f041972054d403dba6ddef5f4971c9614aa0c8f9ba0f054dc7e89289bf8016b")
                message(FATAL_ERROR "The reviewed historical writer source does not match")
            endif()
            foreach(control IN ITEMS historical_writer prk_allocation)
                add_executable(passvault_biometric_windows_${control}_test
                    tests/windows/passvault_biometric_windows_security_test.cpp)
                list(APPEND windows_security_targets passvault_biometric_windows_${control}_test)
            endforeach()
            target_compile_definitions(passvault_biometric_windows_historical_writer_test PRIVATE
                PASSVAULT_BIOMETRIC_PVA036_HISTORICAL_TEST=1
                PASSVAULT_BIOMETRIC_PVA036_SOURCE="${pva036_source}")
            target_compile_definitions(passvault_biometric_windows_prk_allocation_test PRIVATE
                PASSVAULT_BIOMETRIC_PRK_ALLOCATION_TEST=1)
            foreach(file_case IN ITEMS
                success validation_failure dacl_failure collision
                empty_payload oversized_payload empty_suffix rename_failure
            )
                add_test(
                    NAME passvault_biometric_windows_historical_${file_case}
                    COMMAND passvault_biometric_windows_historical_writer_test --file-case ${file_case})
            endforeach()
            foreach(prk_case IN ITEMS unarmed allocation_cut)
                add_test(
                    NAME passvault_biometric_windows_prk_${prk_case}
                    COMMAND passvault_biometric_windows_prk_allocation_test --prk-case ${prk_case})
            endforeach()
        endif()
        # All isolated test images retain the current security target's flags;
        # no observer or replacement allocator is linked into the normal DLL.
        foreach(security_target IN LISTS windows_security_targets)
            target_include_directories(${security_target} PRIVATE include)
            target_compile_definitions(${security_target} PRIVATE
                NOMINMAX
                UNICODE
                _UNICODE
                WIN32_LEAN_AND_MEAN
                WINVER=0x0A00
                _WIN32_WINNT=0x0A00
                PASSVAULT_BIOMETRIC_BUILD=1
            )
            target_compile_options(${security_target} PRIVATE
                /W4
                /WX
                /guard:cf
                /sdl
            )
            target_link_options(${security_target} PRIVATE
                /DYNAMICBASE
                /HIGHENTROPYVA
                /NXCOMPAT
                /guard:cf
            )
            set_property(TARGET ${security_target} PROPERTY
                MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>"
            )
            target_link_libraries(${security_target} PRIVATE
                advapi32
                bcrypt
                ole32
                runtimeobject
                windowsapp
            )
        endforeach()
        add_test(
            NAME passvault_biometric_windows_security
            COMMAND passvault_biometric_windows_security_test
        )
        add_test(
            NAME passvault_biometric_windows_kdf_known_answer
            COMMAND passvault_biometric_windows_security_test --kdf-known-answer
        )
        # New reported-I/O-status controls use real owned files/handles; they
        # do not claim naturally occurring OS faults, crash or cancellation.
        foreach(io_case IN ITEMS write_failure flush_failure close_failure)
            add_test(
                NAME passvault_biometric_windows_file_io_${io_case}
                COMMAND passvault_biometric_windows_security_test --file-io-case ${io_case}
            )
        endforeach()
        # Each failure path is a separate process/result, so a baseline failure
        # cannot make later cases appear to have run. All data is synthetic.
        foreach(file_case IN ITEMS
            success validation_failure dacl_failure collision
            empty_payload oversized_payload empty_suffix rename_failure
        )
            add_test(
                NAME passvault_biometric_windows_file_${file_case}
                COMMAND passvault_biometric_windows_security_test --file-case ${file_case}
            )
        endforeach()
        # Guard-only synthetic checks return before all filesystem/provider tests.
        foreach(secret_case IN ITEMS
            normal_scope early_return allocation_exception nested_exception
        )
            add_test(
                NAME passvault_biometric_windows_secret_${secret_case}
                COMMAND passvault_biometric_windows_security_test --secret-wipe-case ${secret_case}
            )
        endforeach()
    endif()
else()
    message(FATAL_ERROR "The biometric bridge is intentionally unavailable on this platform")
endif()

target_include_directories(passvault_biometric PUBLIC include)
target_compile_definitions(passvault_biometric PRIVATE PASSVAULT_BIOMETRIC_BUILD=1)

if(BUILD_TESTING)
    add_executable(passvault_biometric_abi_test tests/passvault_biometric_abi_test.cpp)
    target_link_libraries(passvault_biometric_abi_test PRIVATE passvault_biometric)
    add_test(NAME passvault_biometric_abi COMMAND passvault_biometric_abi_test)
endif()

install(TARGETS passvault_biometric
    LIBRARY DESTINATION .
    RUNTIME DESTINATION .
)
