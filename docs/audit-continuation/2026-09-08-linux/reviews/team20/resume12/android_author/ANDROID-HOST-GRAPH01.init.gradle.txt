def reject = { String why -> throw new GradleException('AndroidHostGraph01: ' + why) }
def textBound = { Object value, int cap ->
    if (!(value instanceof String) || value.empty || value.length() > cap || !(value ==~ /[ -~]+/)) {
        reject('bounded printable text required')
    }
    value
}
def taskPath = { Object value ->
    String s = textBound(value, 512)
    if (!(s ==~ /:(?:[A-Za-z0-9_-]+:)*[A-Za-z0-9_-]+/)) reject('bounded task path required')
    s
}
def typeName = { Object value ->
    String s = textBound(value, 512)
    if (!(s ==~ /[A-Za-z_$][A-Za-z0-9_$]*(?:\.[A-Za-z_$][A-Za-z0-9_$]*)*/)) reject('bounded type required')
    s
}
def absolutePath = { Object value ->
    String s = textBound(value, 1024)
    def p = new File(s).toPath()
    if (!p.isAbsolute() || p.normalize().toString() != s) reject('absolute normalized path required')
    p
}

// This is a callback-local failure-tree observation, not a final-process oracle.
// Traverse public cause/suppressed/getCauses only; no messages, stack traversal,
// getDeclaredMethod, setAccessible, arbitrary cause adapters or exception imports.
def failureTree = { Throwable failure ->
    def nodes = [], ids = new IdentityHashMap(), active = new IdentityHashMap()
    int edgeCount = 0
    def walk
    walk = { Throwable f ->
        if (f == null || active.containsKey(f)) reject('missing/cyclic failure node')
        if (ids.containsKey(f)) return ids.get(f)
        if (nodes.size() >= 32) reject('failure node bound')
        int id = nodes.size()
        ids.put(f, id)
        active.put(f, true)
        def row = [id: id, type: typeName(f.class.name), intendedIdentity: intended != null && f.is(intended),
                   children: [], suppressedCount: null]
        nodes.add(row)
        // Reject unknown classes BEFORE any virtual/reflected cause getter.
        if (row.intendedIdentity ? row.type != 'PassVaultAndroidHostGraphOnlyRefusal' : !WRAPPERS.contains(row.type)) {
            reject('unrecognized failure type')
        }
        row.suppressedCount = f.suppressed.length
        if (row.suppressedCount != 0) reject('suppressed failure is not an exclusive intended refusal')
        def children = [], unique = new IdentityHashMap()
        def child = { Object value ->
            if (!(value instanceof Throwable)) reject('non-Throwable cause')
            if (!unique.containsKey(value)) { unique.put(value, true); children.add(value) }
        }
        def primary = f.cause
        if (primary != null) child(primary)
        def getter = null
        try { getter = f.class.getMethod('getCauses') } catch (NoSuchMethodException absent) { }
        if (getter != null) {
            def extra = getter.invoke(f)
            if (!(extra instanceof List) || extra.size() > 16) reject('public getCauses bound/type')
            extra.each { child(it) }
        }
        if (row.intendedIdentity) {
            if (!children.empty || row.type != 'PassVaultAndroidHostGraphOnlyRefusal') reject('changed refusal leaf')
        } else if (children.empty) {
            reject('unrelated failure leaf')
        }
        if ((edgeCount += children.size()) > 64) reject('failure edge bound')
        row.children = children.collect { walk(it) }
        active.remove(f)
        id
    }
    try {
        walk(failure)
        if (nodes.count { it.intendedIdentity } != 1) reject('unique intended identity required')
        [complete: true, onlyIntendedAtCallback: true, root: 0, nodes: nodes, errorType: null]
    } catch (Throwable error) {
        [complete: false, onlyIntendedAtCallback: false, root: nodes.empty ? null : 0,
         nodes: nodes, errorType: typeName(error.class.name)]
    }
}

        finishCallbacks: finishCount, beforeTaskEntries: beforeCount, afterTaskEvents: afterCount,
        taskActionsAdmitted: false, localExpectedRefusal: local, failureTree: tree,
        finalProcessAndLogReviewRequired: true, observedTestCases: 0,
        qualification: 'Callback-local tree and task-entry guards; configuration can execute code and write files.'])
}

        if (paths.size() != tasks.size()) reject('duplicate requested graph path')
        int edgeRefs = 0
        def edge = { collection ->
            if (!(collection instanceof Collection) || collection.size() > 128) reject('edge-list bound/type')
            if ((edgeRefs += collection.size()) > 8192) reject('total edge-reference bound')
            def names = collection.collect { taskPath(it.path) }

            names.sort()
        }
        def rows = tasks.collect { t ->
            // Populated execution-graph dependencies, NOT Task.inputs/buildDependencies/classpaths.
            def dependencies = edge(graph.getDependencies(t))
            if (dependencies.any { !paths.contains(it) }) reject('dependency outside populated graph')
            [path: taskPath(t.path), type: typeName(t.class.name), enabled: t.enabled,
             isTest: t instanceof Test, isExec: t instanceof Exec, isJavaExec: t instanceof JavaExec,
             isArchive: t instanceof AbstractArchiveTask, dependencies: dependencies,
