class Refused(Exception):
    pass


def need(ok, code):
    if not ok:
        raise Refused(code)


def cancelled(*unused):
    global CANCEL
    CANCEL = True


signal.signal(signal.SIGINT, cancelled)
signal.signal(signal.SIGTERM, cancelled)


def tick():
    need(not CANCEL and time.monotonic() - START < 95, 'CANCELLED_OR_TOTAL_DEADLINE')


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def pin(st, directory=False):
    out = {k: getattr(st, 'st_' + k) for k in ('dev', 'ino', 'uid', 'mode')}
    if not directory:
        out.update(nlink=st.st_nlink, bytes=st.st_size,
                   mtime_ns=st.st_mtime_ns, ctime_ns=st.st_ctime_ns)
    return out


def directory(path):
    need(path.is_absolute() and '..' not in path.parts, 'DIRECTORY_PATH')
    fd = os.open('/', ODIR)
    try:
        for part in path.parts[1:]:
            new = os.open(part, ODIR, dir_fd=fd)
            os.close(fd)
            fd = new
            st = os.fstat(fd)
            need(st.st_uid == 0 and not st.st_mode & 0o022, 'DIRECTORY_OWNER_MODE')
        out, fd = fd, None
        return out
    finally:
        if fd is not None:
            os.close(fd)


def read_file(path, cap, keep=True):
    parent = directory(path.parent)
    fd = None
    try:
        before = os.stat(path.name, dir_fd=parent, follow_symlinks=False)
        need(stat.S_ISREG(before.st_mode) and before.st_uid == 0 and before.st_nlink == 1
             and not before.st_mode & 0o022 and 0 <= before.st_size <= cap, 'INPUT_TYPE_MODE_SIZE')
        fd = os.open(path.name, os.O_RDONLY | os.O_NOFOLLOW | os.O_NONBLOCK | os.O_CLOEXEC, dir_fd=parent)
        need(pin(before) == pin(os.fstat(fd)), 'INPUT_OPEN_DRIFT')
        h, size, chunks = hashlib.sha256(), 0, []
        while True:
            tick()
            data = os.read(fd, min(65536, cap + 1 - size))
            if not data:
                break
            size += len(data)
            need(size <= cap, 'INPUT_CAP')
            h.update(data)
            if keep:
                chunks.append(data)
        need(size == before.st_size and pin(before) == pin(os.fstat(fd)) ==
             pin(os.stat(path.name, dir_fd=parent, follow_symlinks=False)), 'INPUT_READ_DRIFT')
        return b''.join(chunks), {'pin': pin(before), 'sha256': h.hexdigest()}
    finally:
        try:
            if fd is not None:
                os.close(fd)
        finally:
            os.close(parent)


def unique(pairs):
    out = {}
    for key, value in pairs:
        need(key not in out, 'DUPLICATE_JSON_KEY')
        out[key] = value
    return out


def parse(raw):
    return json.loads(raw.decode('utf-8'), object_pairs_hook=unique,
                      parse_constant=lambda unused: need(False, 'NONFINITE_JSON'))


def save(name, raw):
    global WRITTEN
    need(EFD is not None and re.fullmatch(r'[a-zA-Z0-9.-]+', name), 'EVIDENCE_NAME')
    need(not any(secret in raw for secret in KNOWN_SECRETS), 'KNOWN_CREDENTIAL_ECHO_WITHHELD')
    cap = 2 * MIB if name == 'RESULT.json' else 2 * MIB - 16384
    need(WRITTEN + len(raw) <= cap, 'AGGREGATE_EVIDENCE_CAP')
    fd = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC,
                 0o600, dir_fd=EFD)
    try:
        WRITTEN += len(raw)
        at = 0
        while at < len(raw):
            n = os.write(fd, raw[at:])
            need(n > 0, 'EVIDENCE_SHORT_WRITE')
            at += n
        os.fsync(fd)
    finally:
        os.close(fd)


def record(name, value):
    save(name, (json.dumps(value, ensure_ascii=True, sort_keys=True) + '\n').encode())


def context_guard(point):
    for path in (W.parent, T, T / '.git'):
        fd = directory(path)
        try:
            need(pin(os.fstat(fd), True) == point['images'][str(path)], 'T_DIRECTORY_DRIFT')
        finally:
            os.close(fd)
    unused, image = read_file(T / '.git/config', 4096, keep=False)
    need(image == point['images'][str(T / '.git/config')], 'T_CONFIG_DRIFT')


def tool_guard(tool, expected):
    need(type(tool) is dict and set(tool) == {'path', 'pin', 'sha256'}
         and type(tool['path']) is str, 'TOOL_BINDING')
    path = Path(tool['path'])
    need(path.is_absolute() and path.name == expected, 'TOOL_PATH')
    unused, image = read_file(path, 256 * MIB, keep=False)
    need(image == {'pin': tool['pin'], 'sha256': tool['sha256']}
         and tool['pin']['mode'] & 0o111, 'TOOL_IMAGE_DRIFT')


def query(label, argv, env, tool, cap, documentation=False):
    tick()
    tool_guard(tool, 'curl' if documentation else 'gh')
    row = {'label': label, 'argv': argv, 'cwd': str(T), 'status': 'INTENT',
           'start_unix': time.time(), 'start_elapsed': time.monotonic() - START}
    # Reserve two seconds within the 30s query ceiling for direct kill/wait.
    end = min(START + row['start_elapsed'] + 30, START + 95)
    work_end = end - 2
    RESULT['commands'].append(row)
    record(label + '-INTENT.json', row)
    p, sel, error = None, selectors.DefaultSelector(), None
    buffers, eof = {'stdout': bytearray(), 'stderr': bytearray()}, set()
    limits = {'stdout': cap, 'stderr': 16384}
    try:
        need(not CANCEL and time.monotonic() < work_end, 'QUERY_CANCELLED_OR_DEADLINE')
        p = subprocess.Popen(argv, cwd=T, env=env, stdin=subprocess.DEVNULL,
                             stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                             start_new_session=True, close_fds=True)
        row.update(pid=p.pid, owned_session=p.pid)
        for name, pipe in (('stdout', p.stdout), ('stderr', p.stderr)):
            os.set_blocking(pipe.fileno(), False)
            sel.register(pipe, selectors.EVENT_READ, name)
        while sel.get_map() or p.poll() is None:
            need(not CANCEL and time.monotonic() < work_end, 'QUERY_CANCELLED_OR_DEADLINE')
            for key, unused in sel.select(min(0.1, max(0, work_end - time.monotonic()))):
                name = key.data
                left = limits[name] - len(buffers[name])
                data = os.read(key.fd, min(65536, left + 1))
                if not data:
                    eof.add(name)
                    sel.unregister(key.fileobj)
                    continue
                buffers[name].extend(data[:left])
                need(len(data) <= left, 'QUERY_OUTPUT_CAP')
    except Refused as exc:
        error = str(exc)
    except BaseException:
        error = 'QUERY_SPAWN_OR_OBSERVATION_ERROR'
    finally:
        if p is not None and p.poll() is None:
            try:
                p.kill()  # Exact unreaped Popen child only; never a group/name sweep.
            except BaseException:
                error = 'DIRECT_CHILD_SIGNAL_UNCERTAIN'
        if p is not None:
            try:
                p.wait(timeout=max(0, end - time.monotonic()))
            except BaseException:
                error = 'DIRECT_CHILD_REAP_UNCERTAIN'
            for pipe in (p.stdout, p.stderr):
                try:
                    pipe.close()
                except BaseException:
                    error = 'PIPE_CLOSE_UNCERTAIN'
        try:
            sel.close()
        except BaseException:
            error = 'SELECTOR_CLOSE_UNCERTAIN'
    if time.monotonic() > end:
        error = error or 'QUERY_DEADLINE_EXCEEDED'
    complete = eof == {'stdout', 'stderr'} and error is None
    row.update(exit=None if p is None else p.returncode,
               direct_child_reaped=p is not None and p.returncode is not None,
               complete_streams=complete, end_unix=time.time(),
               elapsed_seconds=time.monotonic() - START - row['start_elapsed'])
    raw, stderr = bytes(buffers['stdout']), bytes(buffers['stderr'])
    if any(secret in raw or secret in stderr for secret in KNOWN_SECRETS):
        error = 'KNOWN_CREDENTIAL_ECHO_WITHHELD'
        row['raw_withheld'] = True
    else:
        row.update(stdout_bytes=len(raw), stdout_sha256=sha(raw),
                   stderr_bytes=len(stderr), stderr_sha256=sha(stderr),
                   hashes_cover='COMPLETE_STREAMS' if complete else 'RETAINED_PREFIXES_ONLY')
        if documentation and complete and p.returncode == 0:
            trailer = re.search(rb'\nPASSVAULT_HTTP_STATUS=([0-9]{3})\n\Z', raw)
            if trailer is None or trailer.start() > MIB:
                error = 'DOCUMENT_STATUS_SUFFIX_INVALID' if trailer is None else 'DOCUMENT_BODY_CAP'
                save(label + '.stdout.partial', raw)
            else:
                body, suffix = raw[:trailer.start()], raw[trailer.start():]
                save('eligibility.html', body)
                row.update(http_status=trailer.group(1).decode('ascii'),
                           stdout_trailer=suffix.decode('ascii'), html_bytes=len(body), html_sha256=sha(body))
                if row['http_status'] != '200':
                    error = 'DOCUMENT_HTTP_STATUS_NOT_200'
        else:
            save(label + ('.stdout' if complete else '.stdout.partial'), raw)
        save(label + '.stderr.private', stderr)
    if not error and (p is None or p.returncode != 0):
        error = 'QUERY_EXIT_NONZERO'
    row.update(status='CAPTURED' if not error else 'FAILED', error=error)
    record(label + '-RESULT.json', row)
    need(error is None and complete and row['direct_child_reaped'], error or 'QUERY_INCOMPLETE')
    tool_guard(tool, 'curl' if documentation else 'gh')
    return raw

def runners_count(raw):
    obj = parse(raw)
    need(type(obj) is dict and set(obj) == {'total_count', 'runners'} and
         type(obj['total_count']) is int and obj['total_count'] >= 0 and
         type(obj['runners']) is list and len(obj['runners']) <= 100, 'RUNNER_LIST_SHAPE')
    ids = set()
    def text(value):
        return type(value) is str and 0 < len(value) <= 256 and all(
            unicodedata.category(c) not in ('Cc', 'Cf', 'Cs') for c in value)
    for r in obj['runners']:
        need(type(r) is dict and set(r) == {'id', 'name', 'os', 'status', 'busy', 'labels'}, 'RUNNER_SHAPE')
        need(type(r['id']) is int and r['id'] > 0 and r['id'] not in ids and type(r['busy']) is bool
             and all(text(r[k]) for k in ('name', 'os', 'status')), 'RUNNER_FIELDS')
        ids.add(r['id'])
        need(type(r['labels']) is list and len(r['labels']) <= 100, 'LABEL_ARRAY')
        for label in r['labels']:
            need(type(label) is dict and set(label) == {'name', 'type'} and
                 all(text(v) for v in label.values()), 'LABEL_FIELDS')
    need(obj['total_count'] <= 100 and len(ids) == obj['total_count'], 'RUNNER_INVENTORY_INCOMPLETE')
    return len(ids)
