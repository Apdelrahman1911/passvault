def checked = { File file, boolean outputPath ->
    def p = file.toPath().toAbsolutePath().normalize()
    if (!(outputPath ? p.startsWith(checkout) : readRoots.any { p.startsWith(new File(it).toPath()) })) reject('path escaped fixed private read/write envelope')
    for (def q = p; q != null; q = q.parent) {
        if (Files.isSymbolicLink(q)) reject('symlink in a selected input/output path')
        if (Files.exists(q, LinkOption.NOFOLLOW_LINKS)) {
            def a = Files.readAttributes(q, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
            if (!a.isDirectory() && !a.isRegularFile()) reject('nonregular selected path/ancestor')
            if (outputPath || q.startsWith(runtime)) {
                if (Files.getAttribute(q, 'unix:uid', LinkOption.NOFOLLOW_LINKS) != 0) reject('nonowned private path')
                if (a.isRegularFile() && Files.getAttribute(q, 'unix:nlink', LinkOption.NOFOLLOW_LINKS) != 1) reject('hardlinked private file')
            }
        }
    }
    p
}
def hashed = { p ->
    def a = Files.readAttributes(p, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
    if (!a.isRegularFile() || a.size() > 256L * 1024 * 1024 || (hashedBytes += a.size()) > 512L * 1024 * 1024) reject('finite regular-file/hash budget exceeded')
    def md = MessageDigest.getInstance('SHA-256')
    byte[] buf = new byte[65536], header = new byte[8]
    long seen = 0L; int copied = 0
    Files.newInputStream(p, LinkOption.NOFOLLOW_LINKS).withCloseable { stream ->
        int n
        while ((n = stream.read(buf)) != -1) {
            if (n == 0) continue
            if ((seen += n) > a.size()) reject('file grew during read')
            int take = Math.min(n, 8 - copied)
            if (take > 0) { System.arraycopy(buf, 0, header, copied, take); copied += take }
            md.update(buf, 0, n)
        }
    }
    def z = Files.readAttributes(p, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
    if (seen != a.size() || z.size() != a.size() || z.lastModifiedTime() != a.lastModifiedTime() || z.fileKey() != a.fileKey()) reject('unstable file identity/read')
    [path: p.toString(), bytes: seen, sha256: md.digest().encodeHex().toString(), header: header.encodeHex().toString()]
}
def members = { List roots, boolean mayBeAbsent ->
    def files = new TreeMap(); int examined = 0
    roots.each { File rootFile ->
        def p = checked(rootFile, true)
        if (!Files.exists(p, LinkOption.NOFOLLOW_LINKS)) {
            if (!mayBeAbsent) reject('required private output absent')
        } else {
            Files.walk(p).withCloseable { stream ->
                def iterator = stream.iterator()
                while (iterator.hasNext()) {
                    def q = checked(iterator.next().toFile(), true)
                    if (++examined > 32768) reject('private output traversal bound')
                    if (!Files.isDirectory(q, LinkOption.NOFOLLOW_LINKS)) files[q.toString()] = q
                }
            }
        }
    }
    files.values().toList()
}

def resourceArchive = { ->
    def files = members(outputs[PACKAGE], false)
    if (files.size() != 1) reject('host resource package must expose exactly one private file')
    def p = files[0]
    if (!p.startsWith(checkout.resolve('app-android/build/intermediates'))) reject('host resource archive outside private intermediates')
    def identity = hashed(p)
    if (!identity.header.startsWith('504b0304')) reject('host resource output must be an ordinary ZIP, not a binary launcher')
    // This small ZIP32 check rejects a recognized APK signing block, ZIP64,
    // appended material and self-extracting prefixes; no authenticity claim.
    new RandomAccessFile(p.toFile(), 'r').withCloseable { f ->
        long n = f.length(); int take = (int)Math.min(n, 65557L)
        byte[] tail = new byte[take]; f.seek(n - take); f.readFully(tail)
        def u16 = { int i -> (tail[i] & 255) | ((tail[i + 1] & 255) << 8) }
        def u32 = { int i -> ((long)(tail[i] & 255)) | ((long)(tail[i + 1] & 255) << 8) | ((long)(tail[i + 2] & 255) << 16) | ((long)(tail[i + 3] & 255) << 24) }
        int eocd = -1
        for (int i = take - 22; i >= 0; --i) {
            if (u32(i) == 0x06054b50L && i + 22 + u16(i + 20) == take) { eocd = i; break }
        }
        if (eocd < 0 || u16(eocd + 4) != 0 || u16(eocd + 6) != 0 || u16(eocd + 8) != u16(eocd + 10) || u16(eocd + 10) > 32768) reject('host resource ZIP32 terminal shape')
        long central = u32(eocd + 16), size = u32(eocd + 12)
        if (central < 4 || central + size != n - take + eocd) reject('host resource ZIP32 directory offset')
        if (central >= 16) {
            byte[] marker = new byte[16]; f.seek(central - 16); f.readFully(marker)
            if (new String(marker, 'US-ASCII') == 'APK Sig Block 42') reject('signed APK block forbidden for host resource scope')
        }
    }
    def names = new HashSet(), essential = [:]; long total = 0L; int count = 0
    new ZipFile(p.toFile()).withCloseable { zip ->
        def entries = zip.entries()
        while (entries.hasMoreElements()) {
            def entry = entries.nextElement(); String name = entry.name
            if (++count > 32768 || !names.add(name) || name.empty || name.length() > 1024 || name.startsWith('/') || name.indexOf(92) >= 0 || name.indexOf(0) >= 0 || name.split('/').any { it == '..' || it == '.' }) reject('unsafe/duplicate host resource ZIP name')
            boolean allowed = name in ['AndroidManifest.xml', 'resources.arsc', 'res/', 'assets/'] || name.startsWith('res/') || name.startsWith('assets/')
            if (!allowed) reject('host resource ZIP contains non-resource/signing/DEX/native-package namespace')
            if (entry.isDirectory()) continue
            if (entry.size < 0 || entry.size > 64L * 1024 * 1024 || (total += entry.size) > 128L * 1024 * 1024) reject('host resource ZIP expansion bound')
            def md = MessageDigest.getInstance('SHA-256'); long seen = 0L; byte[] buffer = new byte[65536]
            zip.getInputStream(entry).withCloseable { stream ->
                int n
                while ((n = stream.read(buffer)) != -1) {
                    if (n == 0) continue
                    if ((seen += n) > entry.size) reject('host resource entry expansion/size mismatch')
                    md.update(buffer, 0, n)
                }
            }
            if (seen != entry.size) reject('host resource entry incomplete')
            if (name in ['AndroidManifest.xml', 'resources.arsc']) essential[name] = [bytes: seen, sha256: md.digest().encodeHex().toString()]
        }
    }
    if (essential.keySet() != ['AndroidManifest.xml', 'resources.arsc'].toSet() || essential.values().any { it.bytes == 0 }) reject('required binary manifest/resource table absent')
    if (hashed(p) != identity) reject('resource archive changed during inspection')
    [identity: identity, entries: count, expandedBytes: total, essential: essential,
     qualification: 'Unsigned resource-only intermediate structure, not app APK/native-loader/install/hardware evidence']
}

def joinConsumer = { Test test ->
    def wanted = [:]
    produced.values().flatten().each { row ->
        if (wanted.containsKey(row.relativeClass)) reject('duplicate critical produced class')
        wanted[row.relativeClass] = row
    }
    if (!wanted.containsKey(CLASS_PACKAGE + 'AndroidClipboardService.class') || !wanted.containsKey(CLASS_PACKAGE + 'AndroidClipboardServiceTest.class')) reject('both actual critical compiler base classes required')

    if (directories.empty || directories.size() > 16) reject('Test class directory bound')
    produced[TEST_COMPILER].each { row ->
        def p = new File(row.path).toPath()
        if (directories.count { p == it.resolve(row.relativeClass) } != 1 || hashed(p).sha256 != row.sha256) reject('fresh compiler class not the exact Test class-directory member')
    }
    def cp = test.classpath.files.toList()

    def found = [:].withDefault { [] }
    cp.collect { checked(it, false) }.unique().each { carrier ->
        if (!Files.exists(carrier, LinkOption.NOFOLLOW_LINKS)) return
        if (Files.isDirectory(carrier, LinkOption.NOFOLLOW_LINKS)) {
            wanted.each { relative, expected ->
                def file = carrier.resolve(relative)
                if (Files.exists(file, LinkOption.NOFOLLOW_LINKS)) {
                    def row = hashed(checked(file.toFile(), false))
                    if (row.sha256 != expected.sha256) reject('directory classpath has a stale critical class')
                    found[relative].add([carrier: carrier.toString(), classSha256: row.sha256])
                }
            }
        } else {
            def before = Files.readAttributes(carrier, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
            if (!before.isRegularFile() || before.size() > 256L * 1024 * 1024) reject('classpath ZIP size/type bound')
            new ZipFile(carrier.toFile()).withCloseable { zip ->
                def entries = zip.entries(); int count = 0; def criticalSeen = new HashSet()
                while (entries.hasMoreElements()) {
                    def entry = entries.nextElement()
                    if (++count > 65536) reject('classpath ZIP entry bound')

                    def md = MessageDigest.getInstance('SHA-256'); long seen = 0L; byte[] buffer = new byte[65536]
                    zip.getInputStream(entry).withCloseable { stream ->
                        int n
                        while ((n = stream.read(buffer)) != -1) {
                            if (n == 0) continue
                            if ((seen += n) > entry.size) reject('classpath class expansion bound')
                            md.update(buffer, 0, n)
                        }
                    }
                    String digest = md.digest().encodeHex().toString()
                    if (seen != entry.size || digest != wanted[entry.name].sha256) reject('runtime carrier class differs from fresh compiler output')
                    found[entry.name].add([carrier: carrier.toString(), classSha256: digest])
                }
            }
            def after = Files.readAttributes(carrier, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
            if (after.fileKey() != before.fileKey() || after.size() != before.size() || after.lastModifiedTime() != before.lastModifiedTime()) reject('classpath carrier changed during selected entry join')
        }
    }
    if (wanted.keySet().any { found[it].size() != 1 }) reject('missing or alternate critical classpath winner')
    emit(common() + [kind: 'compiler-test-join', classDirectories: directories.collect { it.toString() }, classesAndConfig: found,
        resourceConfiguration: configEvidence,
        qualification: 'Fresh declared compiler outputs match the effective configured Test consumer; not internal compiler visitation or hostile-code confinement'])
}

            legal.legalDocuments.files.collect { checked(it, false).toString() }.toSet() != ['LICENSE.txt', 'NOTICE.txt', 'THIRD_PARTY_NOTICES.md'].collect { checkout.resolve(it).toString() }.toSet() ||
            checked(legal.thirdPartyLicenseDirectory.get().asFile, false) != checkout.resolve('THIRD_PARTY_LICENSES')) reject('exact source legal input/Sync deletion boundary')
    if (!members(outputs[LEGAL], true).empty) reject('legal Sync destination must be freshly owned and empty')
    SOURCE_INPUTS.keySet().each { if (!classRows(it, true).empty) reject('critical class exists before compiler actions') }
    schemaEqual('before-actions')
    testPolicy((Test)graph.allTasks.find { it.path == TARGET })
    emit(common() + [kind: 'action-admission', graphDataSha256: EXPECTED_GRAPH_SHA, taskCount: 333, edgeReferences: 663,

        qualification: 'Finite typed actions plus actual generic outputs; cooperative path guards, not arbitrary-plugin sandbox'])

}

beforeAction = { task ->
    phase = 'host-actions'
    if (!control.gate || control.failed || !expectedByPath.containsKey(task.path) || entered.contains(task.path) ||
            task.class.name != expectedByPath[task.path].type || outputPaths(task) != outputs[task.path]) reject('task/output identity drift before entry')
    entered.add(task.path)
    if (task.path == ROOM) schemaEqual('before-schema-finalizer')
    if (task.path == LEGAL && !members(outputs[LEGAL], true).empty) reject('legal Sync acquired foreign output before action')
    if (SOURCE_INPUTS.containsKey(task.path)) {
        if (!classRows(task.path, true).empty) reject('critical compiler output became nonempty before action')
        genericSourceAffinity(task); compilerStarted.add(task.path)
    }
    if (task.path == TARGET) {
        if (control.compilerCompleted.toSet() != SOURCE_INPUTS.keySet() || archiveEvidence == null || !control.schemaFinalizerComplete) reject('required compiled/resource/schema prerequisites not accepted')
        testPolicy((Test)task); schemaEqual('before-test')
        configEvidence = hostConfiguration(); joinConsumer((Test)task)
        emit(common() + [kind: 'host-config-output', evidence: configEvidence,
            qualification: 'Actual declared generated configuration joined to inspected resource archive and effective Test consumer; not instrumentation/component/hardware evidence'])
    }
}
afterAction = { task, state ->
    if (!entered.contains(task.path) || completedTasks.contains(task.path)) reject('unexpected/repeated task outcome')
    completedTasks.add(task.path)
    emit(common() + [kind: 'task-state', task: task.path, type: task.class.name, executed: state.executed,
        skipped: state.skipped, upToDate: state.upToDate, noSource: state.noSource, didWork: state.didWork,
        skipMessage: state.skipMessage, failureType: state.failure?.class?.name])
    if (state.failure != null) return
    if (task.path in [MAIN_COMPILER, TEST_COMPILER, TARGET, ROOM, PACKAGE, CONFIG] &&
            (!state.executed || state.skipped || state.upToDate || state.noSource || state.skipMessage != null ||
                (task.path != ROOM && !state.didWork))) reject('critical Host1 task did not execute fresh successful actions')
    // Schema finalizer must actually execute, but idempotent equality need not
    // rewrite unchanged canonical JSON and therefore need not report didWork.
    if (SOURCE_INPUTS.containsKey(task.path)) {
        def classes = classRows(task.path, false)
        if (!compilerStarted.contains(task.path) || classes.size() > 512 || classes.count { it.relativeClass == CLASS_PACKAGE + SOURCE_INPUTS[task.path].base + '.class' } != 1) reject('fresh critical compiler class cohort missing')
        produced[task.path] = classes; control.compilerCompleted.add(task.path)
        emit(common() + [kind: 'compiler-classes', task: task.path, classes: classes, testCases: 0])
    }
    if (task.path == ROOM) { schemaEqual('after-schema-finalizer'); control.schemaFinalizerComplete = true }
    if (task.path == PACKAGE) { archiveEvidence = resourceArchive(); emit(common() + [kind: 'host-resource-archive', evidence: archiveEvidence]) }
    if (task.path == TARGET) {
        if (control.caseStarts != 1 || control.caseCompletes != 1 || control.casePasses != 1) reject('fresh Test lacks the exact actual case stream')
        control.testComplete = true
    }
}
