#!/usr/bin/python3.12
"""Prospective one-shot three pure projection control groups; no Gradle/runtime."""
import hashlib, json, os, signal, subprocess, time
from pathlib import Path
H = Path(__file__).parent
D = H.parent
R = H / 'CONTROL-RESULT.json'
assert not R.exists()
assert not json.loads((D / 'EXECUTION_SLOT.json').read_text())['operation']['active']
approval = json.loads((H / 'CONTROL-APPROVAL.json').read_text())
assert approval['reviewer'] == '/root/current_ledger'
assert approval['status'] == 'ACCEPT_THREE_IN_MEMORY_CONTROL_GROUPS_ONLY'
for name in ('PROJECTION.py.txt', 'CONTROLS.py.txt', 'RUN_CONTROLS.py.txt'):
    assert hashlib.sha256((H / name).read_bytes()).hexdigest() == approval['sha256'][name]
p = None
result = {'scope': 'THREE_INFRASTRUCTURE_NON_XML_CONTROL_GROUPS_NOT_APP_CASES', 'gradle_stop': 'NOT_APPLICABLE', 'temporary_files': 'NONE_CREATED'}
def cancel(signum, frame):
    raise InterruptedError('ORIGINAL_CONTROL_INTERRUPTED')
for sig in (signal.SIGTERM, signal.SIGINT):
    signal.signal(sig, cancel)
started = time.monotonic()
try:
    p = subprocess.Popen(['/usr/bin/python3.12', '-I', '-S', '-B', str(H / 'CONTROLS.py.txt')],
        cwd=H, env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','LC_ALL':'C.UTF-8','TZ':'UTC'},
        stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.PIPE, start_new_session=True)
    out, err = p.communicate(timeout=30)
    result.update(exit=p.returncode, stdout=out.decode('utf-8'), stderr=err.decode('utf-8'))
    expected = ('projection-control chunk-boundary-crlf-eof PASS\n'
                'projection-control bounded-private-vocabulary PASS\n'
                'projection-control nonauthority-and-admission PASS\n').encode()
    result['passed'] = p.returncode == 0 and out == expected and err == b''
except BaseException as error:
    result.update(passed=False, error=type(error).__name__)
finally:
    if p is not None:
        if p.poll() is None:
            os.killpg(p.pid, signal.SIGTERM)
            try:
                p.wait(timeout=5)
            except subprocess.TimeoutExpired:
                os.killpg(p.pid, signal.SIGKILL)
                p.wait(timeout=5)
        result['exit'] = p.returncode
        try:
            os.killpg(p.pid, 0)
            result['original_group_settled'] = False
        except ProcessLookupError:
            result['original_group_settled'] = True
    result['elapsed_seconds'] = time.monotonic() - started
    R.write_text(json.dumps(result, indent=2) + '\n')
raise SystemExit(0 if result.get('passed') and result.get('original_group_settled') else 1)
