def pin(s,directory=False):
 v={k:getattr(s,'st_'+k) for k in ('dev','ino','uid','mode')}
 if not directory:v.update(nlink=s.st_nlink,bytes=s.st_size,mtime_ns=s.st_mtime_ns,ctime_ns=s.st_ctime_ns)
 return v
def capture(p):

 try:
  with os.fdopen(fd,'rb',closefd=False) as f:b=f.read(32*1024**2+1)
  assert len(b)==before.st_size and pin(before)==pin(os.fstat(fd))==pin(target.lstat()) and pin(original)==pin(p.lstat())
 finally:os.close(fd)
 v={'sha256':hashlib.sha256(b).hexdigest(),'pin':pin(before)}

 return v
