    if (!(outputPath ? p.startsWith(checkout) : readRoots.any { p.startsWith(new File(it).toPath()) })) reject('path escaped fixed private read/write envelope')
    for (def q = p; q != null; q = q.parent) {
        if (Files.isSymbolicLink(q)) reject('symlink in a selected input/output path')
        if (Files.exists(q, LinkOption.NOFOLLOW_LINKS)) {
            def a = Files.readAttributes(q, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
            if (!a.isDirectory() && !a.isRegularFile()) reject('nonregular selected path/ancestor')
            if (outputPath || q.startsWith(runtime)) {
                if (Files.getAttribute(q, 'unix:uid', LinkOption.NOFOLLOW_LINKS) != 0) reject('nonowned private path')
                if (a.isRegularFile() && Files.getAttribute(q, 'unix:nlink', LinkOption.NOFOLLOW_LINKS) != 1) reject('hardlinked private file')
            }
        }
    }
    p
}
def hashed = { p ->
    def a = Files.readAttributes(p, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
    if (!a.isRegularFile() || a.size() > 256L * 1024 * 1024 || (hashedBytes += a.size()) > 512L * 1024 * 1024) reject('finite regular-file/hash budget exceeded')
    def md = MessageDigest.getInstance('SHA-256')
    byte[] buf = new byte[65536], header = new byte[8]
    long seen = 0L; int copied = 0
    Files.newInputStream(p, LinkOption.NOFOLLOW_LINKS).withCloseable { stream ->
        int n
        while ((n = stream.read(buf)) != -1) {
            if (n == 0) continue
            if ((seen += n) > a.size()) reject('file grew during read')
            int take = Math.min(n, 8 - copied)
            if (take > 0) { System.arraycopy(buf, 0, header, copied, take); copied += take }
            md.update(buf, 0, n)
        }
    }
    def z = Files.readAttributes(p, BasicFileAttributes, LinkOption.NOFOLLOW_LINKS)
    if (seen != a.size() || z.size() != a.size() || z.lastModifiedTime() != a.lastModifiedTime() || z.fileKey() != a.fileKey()) reject('unstable file identity/read')
    [path: p.toString(), bytes: seen, sha256: md.digest().encodeHex().toString(), header: header.encodeHex().toString()]
}
def members = { List roots, boolean mayBeAbsent, boolean outputPath ->
    def files = new TreeMap(); int examined = 0
    roots.each { File rootFile ->
        def p = checked(rootFile, outputPath)
        if (!Files.exists(p, LinkOption.NOFOLLOW_LINKS)) {
            if (!mayBeAbsent) reject('required private output absent')
        } else {
            Files.walk(p).withCloseable { stream ->
                def iterator = stream.iterator()
                while (iterator.hasNext()) {
                    def q = checked(iterator.next().toFile(), outputPath)
                    if (++examined > 32768) reject('private output traversal bound')
                    if (!Files.isDirectory(q, LinkOption.NOFOLLOW_LINKS)) files[q.toString()] = q
                }
            }
        }
    }
    files.values().toList()
}
def schemaEqual = { String at ->
    def rows = members([schemaRoot.toFile()], false, true).collect { hashed(it) }
    def actual = rows.collectEntries { [(schemaRoot.relativize(new File(it.path).toPath()).toString()): it.sha256] }
    if (actual != SCHEMAS) reject('runtime-copy canonical schema set/bytes drift')
    emit(common() + [kind: 'schema-equality', at: at, root: schemaRoot.toString(), files: rows, equal: true])
}
def outputPaths = { task ->
    def roots = task.outputs.files.files.toList()
    if (roots.size() > 64) reject('per-task output-root bound')
    def module = checkout.resolve(task.project.path.substring(1).replace(':', '/'))
    def build = module.resolve('build')
    def paths = roots.collect { checked(it, true) }.unique().sort { it.toString() }
    paths.each { p ->
        if (task.path == ROOM) {
            if (p != schemaRoot) reject('Room finalizer output must be only the original private copied schemas')
        } else if (p == build || !p.startsWith(build) || p.startsWith(build.resolve('outputs/apk')) || p.startsWith(build.resolve('outputs/bundle'))) {
            reject('output is not an allowed module build intermediate/report')
        }
    }
    if (task.path == ROOM) {
        // Room2.8.4 marks this real writer destination @Internal, not @OutputDirectory.
        // Keep generic roots as a separate witness; never manufacture a declared output.
        if (task.class.superclass?.name != 'androidx.room.gradle.RoomSchemaCopyTask') reject('Room writer class hierarchy changed')
        def directory = task.getSchemaDirectory()
        if (!(directory instanceof DirectoryProperty) || !directory.isPresent()) reject('Room writer destination property is absent or changed')
        def destination = checked(directory.get().asFile, true)
        if (destination != schemaRoot) reject('Room writer destination must be only the original private copied schemas')
        def observed = [api: 'androidx.room.gradle.RoomSchemaCopyTask.getSchemaDirectory',
            destination: destination.toString(), genericOutputs: paths.collect { it.toString() }]
        if (roomOutputBinding != null && roomOutputBinding != observed) reject('Room writer/generic-output binding drift before action')
        roomOutputBinding = observed
        paths = [destination]
    }
    if (task.path in (SELECTORS + [LEGAL]) && paths.empty) reject('required producer has no bound generic output')
    paths.collect { it.toFile() }
}

        requiredRole('CONFIGURATION_OUTPUT_NOT_RULE_INPUTS')
    } else {
        def nativeFiles = requiredRole('MERGED_NATIVE_DECLARED_DIRECTORY').files.findAll { it.path.endsWith('.so') }
        if (nativeFiles.empty || nativeFiles.any { !it.header.startsWith('7f454c46') }) reject('fresh native merge lacks ELF shared-library bytes')
        def inputsByHash = inputEvidence[NATIVE].files.groupBy { it.sha256 }
        byteMatches = nativeFiles.collect { file ->
            [output: file.path, sha256: file.sha256,
             sameByteInputCandidates: (inputsByHash[file.sha256] ?: []).collect { it.path }.sort()]
        }
    }
    [task: task.path, roles: rows, nativeSameByteCandidates: byteMatches,
     qualification: 'Fresh successful task plus actual declared-output bytes/hashes; native hash matches are candidate origins, not unique variant provenance; DEX headers are not retention/disassembly proof.']
}

gradle.taskGraph.whenReady { graph ->
    readyCount++
    phase = 'whole-graph-barrier'
    try {
        if (!bindingOk || nativeDenied.get() || readyCount != 1 || start.dryRun || control.gate) reject('one bound pre-action callback required')
        def prior = controlData(EXPECTED_GRAPH_PATH, 226997, EXPECTED_GRAPH_SHA)
        def eligibility = controlData(ELIGIBILITY_PATH, 34438, ELIGIBILITY_SHA)
        if (prior.sourceManifestSha256 != SOURCE_SHA || prior.subjectSha256 != SUBJECT_SHA || prior.graphOnly != true ||
                eligibility.graphSha256 != EXPECTED_GRAPH_SHA || eligibility.selectors != SELECTORS ||
                eligibility.taskActionsAdmitted != false) reject('source eligibility basis mismatch')
        expectedByPath = prior.tasks.collectEntries { [(it.path): it] }
        def eligible = eligibility.eligibleTaskPaths.toSet()
        def tasks = graph.allTasks
        def paths = tasks.collect { taskPath(it.path) }.toSet()
        if (tasks.empty || paths.size() != tasks.size() || !eligible.containsAll(paths) ||
                !paths.containsAll(SELECTORS + [ROOM]) || eligibility.forbiddenObservedFinalTasks.any { paths.contains(it) }) {
            reject('unknown/out-of-scope/forbidden actual graph; no task skipping or expansion')
        }
        int edgeRefs = 0
        def edge = { collection ->
            if (!(collection instanceof Collection) || collection.size() > 128) reject('edge-list bound/type')
            if ((edgeRefs += collection.size()) > 8192) reject('edge-reference bound')
            def names = collection.collect { taskPath(it.path) }
            if (names.toSet().size() != names.size()) reject('duplicate edge')
            names.sort()
        }
        def rows = tasks.collect { t ->
            def dependencies = edge(graph.getDependencies(t))
            if (dependencies.any { !paths.contains(it) }) reject('dependency outside actual graph')
            [path: taskPath(t.path), type: typeName(t.class.name), enabled: t.enabled,
             isTest: t instanceof Test, isExec: t instanceof Exec, isJavaExec: t instanceof JavaExec,
             isArchive: t instanceof AbstractArchiveTask, dependencies: dependencies,
             finalizers: edge(t.finalizedBy.getDependencies(t)), mustRunAfter: edge(t.mustRunAfter.getDependencies(t)),
             shouldRunAfter: edge(t.shouldRunAfter.getDependencies(t))]
        }
        if (rows.any { it != expectedByPath[it.path] || !it.enabled || it.isTest || it.isExec || it.isJavaExec ||
                it.finalizers.any { finalizer -> !paths.contains(finalizer) } }) reject('actual typed graph/edges changed')
        // isArchive and package/bundle/publish name substrings are NOT denials.
        // The reviewed finite candidate includes ordinary JAR/local-lint AAR roles.
        if (gradle.rootProject.projectDir.toPath().toAbsolutePath().normalize() != checkout) reject('original copied checkout mismatch')
        tasksByPath = tasks.collectEntries { [(it.path): it] }
        actualByPath = rows.collectEntries { [(it.path): it] }
        tasks.each { task ->
            outputs[task.path] = outputPaths(task)
            outputRows.add([task: task.path, type: task.class.name,
                outputs: outputs[task.path].collect { it.toString() },
                role: task.path == ROOM ? 'PRIVATE_SOURCE_COPY_SCHEMA_EXCEPTION' : 'PRIVATE_MODULE_BUILD_INTERMEDIATE_OR_REPORT'])
        }
        // LEGAL is neither selected nor in the unchanged finite eligibility.
        // Keep its legacy boundary conditional, not a hidden third root or
        // a completion requirement. This attempt supplies no legal/package validation.
        if (tasksByPath.containsKey(LEGAL)) {
            def legal = tasksByPath[LEGAL]
            def legalRoot = checkout.resolve('app-android/build/generated/assets/prepareAndroidLegalAssets')
            if (legal.class.name != 'Build_gradle$PrepareAndroidLegalAssets_Decorated' ||
                    legal.class.superclass?.name != 'Build_gradle$PrepareAndroidLegalAssets' ||
                    checked(legal.outputDirectory.get().asFile, true) != legalRoot ||
                    outputs[LEGAL].collect { it.toPath() } != [legalRoot] ||
                    legal.legalDocuments.files.collect { checked(it, false).toString() }.toSet() !=
                        ['LICENSE.txt', 'NOTICE.txt', 'THIRD_PARTY_NOTICES.md'].collect { checkout.resolve(it).toString() }.toSet() ||
                    checked(legal.thirdPartyLicenseDirectory.get().asFile, false) != checkout.resolve('THIRD_PARTY_LICENSES')) reject('legal Sync source/deletion boundary')
            if (!members(outputs[LEGAL], true, true).empty) reject('legal Sync destination is not fresh')
        }
        producerContract = selectedContract()
        producerContract.declaredOutputs.findAll { it.present }.each { role ->
            if (!members([new File(role.path)], true, true).empty) reject('selected producer output already has bytes before actions')
        }
        schemaEqual('before-actions')
        emit(common() + [kind: 'action-admission', graphDataSha256: EXPECTED_GRAPH_SHA,
            eligibilitySha256: ELIGIBILITY_SHA, taskCount: tasks.size(), edgeReferences: edgeRefs,
            tasks: rows, outputRoots: outputRows, producerContract: producerContract,
            roomDestinationBinding: roomOutputBinding, taskCountIsActualNotForecast: true,
            qualification: 'Whole actual graph checked against finite reviewed source eligibility, including nonpublishing intermediates; cooperative declared-output/Room guard, not arbitrary-plugin sandbox.'])
        control.graphPaths = paths.toList().sort()
        graphCount++; control.actionsAdmitted = true; control.gate = true
    } catch (Throwable error) { control.failed = true; control.gate = false; throw error }
}
beforeAction = { task ->
    phase = 'producer-actions'
    if (!control.gate || control.failed || !actualByPath.containsKey(task.path) || entered.contains(task.path) ||
            !task.enabled || task.class.name != actualByPath[task.path].type || outputPaths(task) != outputs[task.path]) reject('task/output identity drift at entry')
    entered.add(task.path)
    if (task.path == ROOM) schemaEqual('before-schema-finalizer')
    if (task.path == LEGAL && !members(outputs[LEGAL], true, true).empty) reject('legal Sync acquired foreign outputs')
    if (task.path in SELECTORS) {
        if (selectedContract() != producerContract) reject('selected producer API/declared-path binding drift')
        // Do not invent an ordering edge from Room to the two selected producers.
        // Its destination/integrity are guarded at graph/entry/outcome; final
        // success still requires the retained finalizer's accepted completion.
        producerContract.declaredOutputs.findAll { it.task == task.path && it.present }.each { role ->
            if (!members([new File(role.path)], true, true).empty) reject('selected output acquired bytes before producer action')
        }
        inputEvidence[task.path] = producerInputs(task)
        emit(common() + [kind: 'producer-inputs', evidence: inputEvidence[task.path]])
        control.inputCompletions.add(task.path)
    }
}
afterAction = { task, state ->
    if (!entered.contains(task.path) || completedTasks.contains(task.path)) reject('unexpected/repeated outcome')
    completedTasks.add(task.path)
    emit(common() + [kind: 'task-state', task: task.path, type: task.class.name, executed: state.executed,
        skipped: state.skipped, upToDate: state.upToDate, noSource: state.noSource, didWork: state.didWork,
        skipMessage: state.skipMessage, failureType: state.failure?.class?.name])
    if (state.failure != null) return
    if (task.path == ROOM) {
        boolean action = state.executed && !state.skipped && !state.upToDate && !state.noSource && state.skipMessage == null
        boolean unchangedNoSource = state.executed && state.skipped && !state.upToDate && state.noSource &&
            !state.didWork && state.skipMessage == 'NO-SOURCE'
        if (!action && !unchangedNoSource) reject('Room requires successful action or exact unchanged NO-SOURCE')
        schemaEqual('after-schema-finalizer')
        control.schemaFinalizerComplete = true
    }
    if (task.path in SELECTORS) {
        if (!state.executed || state.skipped || state.upToDate || state.noSource || state.skipMessage != null || !state.didWork) reject('selected producer did not execute fresh successful work')
        if (!control.inputCompletions.contains(task.path) || selectedContract() != producerContract) reject('missing input/changed output coupling')
        producedEvidence[task.path] = producerOutputs(task)
        emit(common() + [kind: 'producer-outputs', evidence: producedEvidence[task.path]])
        control.outputCompletions.add(task.path)
    }
}
