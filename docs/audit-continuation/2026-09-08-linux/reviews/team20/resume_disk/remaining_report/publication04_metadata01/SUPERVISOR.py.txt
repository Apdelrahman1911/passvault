def need(ok, message):
    if not ok: raise RuntimeError(message)
def error(exc):
    if len(ERR) < 16 and str(exc)[:240] not in ERR: ERR.append(str(exc)[:240])
def interrupted(number, _frame):
    if len(CANCEL) < 16: CANCEL.append(number)
    if DONE: raise SystemExit(128+number)
def identity(s): return [s.st_dev, s.st_ino, s.st_uid, stat.S_IMODE(s.st_mode)]
def pin(s):
    return dict(dev=s.st_dev, ino=s.st_ino, uid=s.st_uid, mode=s.st_mode, nlink=s.st_nlink,
                bytes=s.st_size, mtime_ns=s.st_mtime_ns, ctime_ns=s.st_ctime_ns)
def encoded(value): return (json.dumps(value, sort_keys=True, separators=(",", ":"))+"\n").encode()
def object_id(kind, data): return hashlib.sha1(kind+b" "+str(len(data)).encode()+b"\0"+data).hexdigest()
def pinned(spec):
    with os.fdopen(os.open(spec["path"], os.O_RDONLY|os.O_NOFOLLOW|os.O_NONBLOCK), "rb") as f:
        s = os.fstat(f.fileno()); need(stat.S_ISREG(s.st_mode) and s.st_uid == 0 and s.st_nlink == 1 and not s.st_mode & 0o022, "unsafe input")
        need(pin(s) == spec["pin"] and s.st_size <= 16*1024**2, "input pin/cap")
        data = f.read(16*1024**2+1)
        need(len(data) == s.st_size and hashlib.sha256(data).hexdigest() == spec["sha256"] and pin(os.fstat(f.fileno())) == pin(os.stat(spec["path"], follow_symlinks=False)) == spec["pin"], "input drift")
def guard(work=True, entry=False):

    with open("/proc/meminfo", "rb") as f: raw = f.read(16385)

    mem = {p[0]: int(p[1])*1024 for p in map(bytes.split, raw.splitlines()) if len(p) == 3 and p[0] in (b"MemAvailable:", b"MemTotal:") and p[2] == b"kB"}
    available, total = mem[b"MemAvailable:"], mem[b"MemTotal:"]; need(0 < available <= total, "meminfo values")
    OBS.update(current_mem_available_bytes=available, minimum_mem_available_bytes=min(OBS.get("minimum_mem_available_bytes", available), available), mem_total_bytes=total)
    need(available >= (12 if entry else 8)*1024**3 and available*(4 if entry else 5) >= total, "RAM floor")
    for key in ("repo", "evidence", "coordinator_parent"):
        need(identity(os.fstat(FDS[key])) == identity(os.stat(request[key]["path"], follow_symlinks=False)) == request[key]["identity"], "original directory drift")
        v = os.fstatvfs(FDS[key]); free = v.f_bavail*v.f_frsize

        need(free >= 3*1024**3, "3GiB disk floor")
    need(pin(os.fstat(FDS["coordinator"])) == pin(os.stat(os.path.basename(request["coordinator"]["path"]), dir_fd=FDS["coordinator_parent"], follow_symlinks=False)) == request["coordinator"]["pin"], "coordinator drift")
    for key in ("source", "python", "git"):

def save(name, data, final=False):
    global USED
    need(USED+len(data) <= CAP-(0 if final else RESERVE), "aggregate evidence cap"); USED += len(data)
    fd = os.open(name, os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW, 0o600, dir_fd=FDS["evidence"])
    with os.fdopen(fd, "wb") as f:
        need(f.write(data) == len(data), "short evidence write"); f.flush(); os.fsync(f.fileno())
    os.fsync(FDS["evidence"]); LOGS[name] = dict(bytes=len(data), sha256=hashlib.sha256(data).hexdigest())
def command(label, args, payload=b""):

    COMMANDS.append(row); deadline = min(WORK, time.monotonic()+20)
    try:
        guard(); need(not ERR and len(payload) <= 4096, "command admission/stdin cap")

        p = subprocess.Popen(row["argv"], cwd=REPO, env=ENV, stdin=subprocess.PIPE if payload else subprocess.DEVNULL,
                             stdout=subprocess.PIPE, stderr=subprocess.PIPE, close_fds=True, start_new_session=True)
        row["pid"] = p.pid; pfd = os.pidfd_open(p.pid, 0)
    except Exception as exc: error(exc)
    finally:
        if p is not None:
            if p.stdin is not None:
                f, p.stdin = p.stdin, None
                try:
                    if pfd is not None and not ERR and not CANCEL:
                        need(time.monotonic() < deadline and len(payload) <= os.fpathconf(f.fileno(), "PC_PIPE_BUF"), "bounded stdin admission")
                        os.set_blocking(f.fileno(), False); need(os.write(f.fileno(), payload) == len(payload), "short bounded stdin")
                except Exception as exc: error(exc)
                finally:
                    try: f.close()
                    except Exception as exc: error(exc)
            for i, f in enumerate((p.stdout, p.stderr)):
                try: os.set_blocking(f.fileno(), False); pipes[i] = f
                except Exception as exc:
                    error(exc)
                    try: f.close()
                    except Exception as exc: error(exc)
            while pfd is not None and time.monotonic() < END-3:
                now = time.monotonic()
                if now >= deadline: error("command deadline")
                if CANCEL or ERR:
                    sig = None
                    if term is None: term = now; sig = signal.SIGTERM
                    elif now-term >= 5 and not killed: killed = True; sig = signal.SIGKILL
                    if sig is not None:
                        try: signal.pidfd_send_signal(pfd, sig)
                        except ProcessLookupError: pass
                        except Exception as exc: error(exc)
                    if now-term >= 15: break
                try:
                    for i, f in list(pipes.items()):
                        try: data = os.read(f.fileno(), 65536)
                        except BlockingIOError: continue
                        if not data: del pipes[i]; f.close(); eof.add(i); continue
                        keep = data[:max(0, CAP-RESERVE-USED-sum(map(len, buffers)))]; buffers[i].extend(keep)
                        need(len(keep) == len(data), "command aggregate output cap")
                    dead = os.waitid(os.P_PIDFD, pfd, os.WEXITED|os.WNOHANG|os.WNOWAIT)
                    if dead is not None and len(eof) == 2:
                        row["exit"] = p.wait(timeout=0); row["reaped"] = True; break
                    if not ERR and not CANCEL: guard()
                except Exception as exc: error(exc)
                time.sleep(0.02)
            for f in pipes.values():
                try: f.close()
                except Exception as exc: error(exc)
            if pfd is not None and not row["reaped"] and not killed:
                killed = True
                try: signal.pidfd_send_signal(pfd, signal.SIGKILL)
                except ProcessLookupError: pass
                except Exception as exc: error(exc)
            if not row["reaped"]:
                try: row["exit"] = p.wait(timeout=max(0, min(1, END-time.monotonic()))); row["reaped"] = True
                except Exception as exc: error("original child settlement unknown: "+type(exc).__name__)
            if pfd is not None:
                try: os.close(pfd)
                except Exception as exc: error(exc)
        row.update(pipe_eof=len(eof), original_pidfd_captured=pfd is not None, term_attempted=term is not None, kill_attempted=killed)
        for suffix, data in zip(("stdout.bin", "stderr.bin"), buffers):
            try: save(label+"-"+suffix, bytes(data))
            except Exception as exc: error(exc)
    need(row["reaped"] and len(eof) == 2 and row["exit"] == 0 and not ERR and not CANCEL, "metadata child failed/HOLD")
    guard(); row["data_complete"] = True; return bytes(buffers[0])
def tree_id(node):
    data = bytearray()
    for name, value in sorted(node.items(), key=lambda row: row[0]+(b"/" if isinstance(row[1], dict) else b"\0")):
        mode, oid = (b"40000", tree_id(value)) if isinstance(value, dict) else value
        data.extend(mode+b" "+name+b"\0"+bytes.fromhex(oid))
    return object_id(b"tree", data)
