def require(value, message):
    if not value:
        raise RuntimeError(message)


def decode(raw):
    def pairs(rows):
        value = {}
        for key, item in rows:
            require(key not in value, 'duplicate JSON key')
            value[key] = item
        return value
    return json.loads(raw, object_pairs_hook=pairs,
                      parse_constant=lambda _: require(False, 'nonfinite JSON'))


def bounded_text(value):
    return (isinstance(value, str) and 0 < len(value) <= 1024
            and all(ord(char) >= 32 and ord(char) != 127 for char in value))

def problem(error):
    return {'type': type(error).__name__, 'errno': getattr(error, 'errno', None),
            'message': str(error)[:256]}


def on_signal(number, _frame):
    global CANCELLED
    if CANCELLED is None:

def check():
    require(CANCELLED is None, 'observer interrupted')
    require(time.monotonic() < END, 'observer overall deadline')


def pin(info, directory=False):
    value = {key: getattr(info, 'st_' + key) for key in ('dev', 'ino', 'uid', 'mode', 'nlink')}
    if not directory:
        value.update(bytes=info.st_size, mtime_ns=info.st_mtime_ns, ctime_ns=info.st_ctime_ns)
    return value


def same_directory(left, right):
    return all(left[key] == right[key] for key in ('dev', 'ino', 'uid', 'mode'))


def directory(path):
    check()
    if path not in DIRS:
        parent = None if path == Path('/') else directory(path.parent)
        require(len(DIRS) < 32, 'fixed parent depth bound')
        fd = os.open('/' if parent is None else path.name, ODIR, dir_fd=parent)
        try:
            value = pin(os.fstat(fd), True)
            require(stat.S_ISDIR(value['mode']) and value['uid'] == 0 and not value['mode'] & 0o022,
                    'parent directory type/owner/mode')
            DIRS[path] = (fd, value)
        except BaseException:
            os.close(fd)
            raise
    return DIRS[path][0]


def recheck_directories():
    check()
    for path, (fd, original) in DIRS.items():
        current = pin(os.fstat(fd), True)
        parent = None if path == Path('/') else DIRS[path.parent][0]
        named = pin(os.stat('/' if parent is None else path.name,
                            dir_fd=parent, follow_symlinks=False), True)
        require(same_directory(current, original) and same_directory(named, original),
                'fresh parent descriptor/name changed')
    return {str(path): pin(os.fstat(DIRS[path][0]), True) for path in PARENTS}


def lock_pin(fd):

    named = pin(os.stat(LOCK.name, dir_fd=DIRS[LOCK.parent][0], follow_symlinks=False))
    require(actual == named == EXPECTED_LOCK, 'original lock pin mismatch; never recreate')
    return actual



def bounded_proc(path, cap):
    check()
    fd = os.open(path, OREAD)
    try:
        require(stat.S_ISREG(os.fstat(fd).st_mode), 'unexpected proc file type')
        data = bytearray()
        while True:
            check()
            chunk = os.read(fd, min(4096, cap - len(data) + 1))
            if not chunk:
                return bytes(data)
            data.extend(chunk)
            require(len(data) <= cap, 'proc byte bound')
    finally:
        os.close(fd)


def resources():
    fields = {}
    for line in bounded_proc('/proc/meminfo', 65536).decode('ascii').splitlines():
        key, value = line.split(':', 1)
        require(key not in fields, 'duplicate meminfo key')
        fields[key] = value.split()
    values = {}
    for key in ('MemAvailable', 'MemTotal'):
        row = fields[key]
        require(len(row) == 2 and row[1] == 'kB' and row[0].isdecimal(), 'meminfo unit/value')
        values[key] = int(row[0]) * 1024
    disks = {}
    for path in PARENTS[:2]:
        point = os.fstatvfs(DIRS[path][0])
        require(point.f_bavail >= 0 and point.f_frsize > 0, 'invalid filesystem headroom')
        disks[str(path)] = point.f_bavail * point.f_frsize
    RESULT['resources'] = {'elapsed_seconds': time.monotonic() - START,
                           'disk_available_bytes': disks, **values}
    require(0 <= values['MemAvailable'] <= values['MemTotal'] and values['MemTotal'] > 0,
            'invalid memory quantities')
    # Root approved disk observations only for this cheap read-only facts batch.
    # Actual 12GiB/authorized-3GiB publication admission remains separate.
    require(values['MemAvailable'] * 4 >= values['MemTotal'], 'metadata RAM25percent floor')

def executable(path):
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        before = os.fstat(fd)
        need(stat.S_ISREG(before.st_mode) and before.st_uid == 0
             and before.st_mode & 0o111 and not before.st_mode & 0o022
             and before.st_nlink == 1 and before.st_size <= 64 * 1024 * 1024, 'executable bounds')
        digest = hashlib.sha256()
        count = 0
        while True:
            data = os.read(fd, 65536)
            if not data:
                break
            count += len(data)
            need(count <= 64 * 1024 * 1024 and time.monotonic() - START < 60 and not CANCELLED,
                 'executable read bound')
            digest.update(data)
        need(pin(before) == pin(os.fstat(fd)) == pin(os.stat(path, follow_symlinks=False))
             and count == before.st_size, 'executable changed')
        return {'path': path, 'pin': pin(before), 'sha256': digest.hexdigest()}
    finally:
        os.close(fd)


def save(name, value):
    data = (json.dumps(value, sort_keys=True, indent=2, allow_nan=False) + '\n').encode()
    need(len(data) <= 65536, 'receipt cap')
    fd = os.open(D / name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
    try:
        with os.fdopen(fd, 'wb', closefd=False) as out:
            out.write(data)
            out.flush()
            os.fsync(fd)
    finally:
        os.close(fd)
    need((D / name).read_bytes() == data, 'receipt readback')


def slot_point():
    """Read only the sole live scheduling record; frozen selected slot is untouched."""
    check()
    fd = os.open(LIVE_SLOT.name, OREAD, dir_fd=directory(LIVE_SLOT.parent))
    try:
        original = pin(os.fstat(fd))
        need(stat.S_ISREG(original['mode']) and original['uid'] == 0
             and original['nlink'] == 1 and not original['mode'] & 0o022
             and original['bytes'] <= 262144, 'live slot bounds')
        raw = bytearray()
        while True:
            check()
            chunk = os.read(fd, min(65536, 262145 - len(raw)))
            if not chunk:
                break
            raw.extend(chunk)
            need(len(raw) <= 262144, 'live slot byte cap')
        need(pin(os.fstat(fd)) == original
             == pin(os.stat(LIVE_SLOT.name, dir_fd=DIRS[LIVE_SLOT.parent][0], follow_symlinks=False))
             and len(raw) == original['bytes'], 'live slot drift')
        value = decode(bytes(raw).decode('utf-8'))
        need(isinstance(value, dict) and isinstance(value.get('state'), str)
             and 0 < len(value['state']) <= 256, 'live slot state projection')
        need(hashlib.sha256(raw).hexdigest() == EXPECTED_SLOT_SHA256
             and value['owner'] == '/root'
             and value['authority'] == 'SOLE_LIVE_POST_FREEZE_SCHEDULING_RECORD_FOR_NEW_OPERATIONS'

             and value['current_operation']['no_auto_chain'] is True
             and value['current_operation']['build_test_CI_admitted'] is False,
             'wrong/stale sole preflight reservation; no U admission')

                     'queue count/projection incomplete or saturated')
                ids = set()
                for run in value['workflow_runs']:
                    need(isinstance(run, dict) and set(run) == {'id', 'status', 'head_sha', 'path', 'head_branch'}
                         and type(run['id']) is int and 0 < run['id'] < 2**64
                         and run['id'] not in ids and run['status'] == name
                         and isinstance(run['head_sha'], str)
                         and re.fullmatch(r'[0-9a-f]{40}', run['head_sha']) is not None
                         and bounded_text(run['path'])
                         and (run['head_branch'] is None or bounded_text(run['head_branch'])),
                         'queue run identity/status/duplicate')
                    ids.add(run['id'])
                RESULT.setdefault('queues', {})[name] = value
                if value['total_count']:
                    RESULT['conflicts'].append({'kind': 'NONTERMINAL_REPOSITORY_CI',
                                                'status': name, 'count': value['total_count']})
            row['status'] = 'POINT_RESPONSE_VALIDATED_NOT_ADMISSION'
            check()
        need(len(RESULT['commands']) == 7 and set(RESULT.get('queues', {})) == set(STATUSES),
             'incomplete fixed batch')
        neutral_parent()
        resources()
        RESULT['live_slot_after'] = slot_point()
        need(RESULT['live_slot_after'] == RESULT['live_slot_before'], 'live slot changed during point')
        RESULT['parents_after'] = recheck_directories()
        RESULT['lock']['after'] = lock_pin(lock_fd)
        check()
        complete = True
    except BaseException as error:
        RESULT['errors'].append(problem(error))
    finally:
        if lock_fd is not None:
            try:
                os.close(lock_fd)  # Point custody only, not custody across a later publication gap.
                RESULT['lock']['close_returned'] = True
            except BaseException as error:
                RESULT['errors'].append(problem(error))
        for fd, _original in reversed(tuple(DIRS.values())):
            try:
                os.close(fd)
            except BaseException as error:
                RESULT['errors'].append(problem(error))
    RESULT['signal'] = CANCELLED
    RESULT['elapsed_seconds'] = round(time.monotonic() - START, 6)
    RESULT['commands_unstarted'] = [name for name, _argv in COMMANDS
                                    if name not in {row['name'] for row in RESULT['commands']}]
    RESULT['cleanup'] = 'Only original command children/pipes/selectors and descriptor-held point parents/lock closed. No unrelated signals or background workers. Abnormal descendants/SIGKILL/kernel blocking are not proven settled. No build/runtime/cache/service/output tree created; no Gradle stop/deletion. Preserve original external terminal completion separately.'
    if complete and not RESULT['errors'] and CANCELLED is None and time.monotonic() < END:
        RESULT['status'] = ('POINT_CONFLICTS_REQUIRE_REASSESSMENT_NOT_ADMISSION' if RESULT['conflicts']
                            else 'POINT_NO_OBSERVED_CONFLICT_NOT_ADMISSION')
        code = 0
    save('PREFLIGHT-01-RESULT.json', RESULT)
    # Receipt is preterminal; a latched late signal/deadline cannot become exit0.
    if CANCELLED is not None or time.monotonic() >= END:
        code = 1
        RESULT['status'] = 'HOLD_CONSUMED_NO_AUTOMATIC_RETRY'
    print(json.dumps({'status': RESULT['status'], 'pending_code': code,
                      'commands_started': len(RESULT['commands']),
                      'commands_reaped': sum(row['child_reaped'] for row in RESULT['commands']),
                      'error_count': len(RESULT['errors']),
                      'elapsed_seconds': RESULT['elapsed_seconds']}))
    return 1 if CANCELLED is not None or time.monotonic() >= END else code
