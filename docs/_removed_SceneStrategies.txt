import androidx.navigation3.runtime.NavEntry
import androidx.navigation3.scene.Scene
import androidx.navigation3.scene.SceneStrategy as NavigationSceneStrategy
import androidx.navigation3.scene.SceneStrategyScope
import androidx.navigation3.scene.SinglePaneSceneStrategy

/**
 * Compatibility helpers for the stable Navigation 3 scene API.
 *
 * Navigation 3 now calculates scenes instead of asking strategies to render
 * composable content directly. These strategies keep the existing PassVault
 * API while delegating rendering to the stable single-pane implementation.
 */
object SceneStrategies {
    fun chain(
        vararg strategies: NavigationSceneStrategy<PassVaultRoute>,
    ): NavigationSceneStrategy<PassVaultRoute> {
        return ChainedSceneStrategy(strategies.toList())
    }
}

private fun singlePaneScene(
    scope: SceneStrategyScope<PassVaultRoute>,
    entries: List<NavEntry<PassVaultRoute>>,
): Scene<PassVaultRoute>? {
    if (entries.isEmpty()) {
        return null
    }

    return with(SinglePaneSceneStrategy<PassVaultRoute>()) {
        scope.calculateScene(entries)
    }
}

private class ChainedSceneStrategy(
    private val strategies: List<NavigationSceneStrategy<PassVaultRoute>>,
) : NavigationSceneStrategy<PassVaultRoute> {
    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? {
        for (strategy in strategies) {
            val scene = with(strategy) {
                this@calculateScene.calculateScene(entries)
            }

            if (scene != null) {
                return scene
            }
        }

        return null
    }
}

/**
 * Auth entries currently use the default Navigation 3 scene.
 */
class VaultAuthSceneStrategy : NavigationSceneStrategy<PassVaultRoute> {
    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? = null
}

/**
 * Selects modal entries tagged as dialogs.
 */
class BlurDialogSceneStrategy : NavigationSceneStrategy<PassVaultRoute> {
    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? {
        return if (entries.lastOrNull()?.isDialogEntry() == true) {
            singlePaneScene(this, entries)
        } else {
            null
        }
    }
}

/**
 * Selects modal entries tagged as bottom sheets.
 */
class BottomSheetSceneStrategy : NavigationSceneStrategy<PassVaultRoute> {
    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? {
        return if (entries.lastOrNull()?.isBottomSheetEntry() == true) {
            singlePaneScene(this, entries)
        } else {
            null
        }
    }
}

/**
 * Keeps the existing split-view hook while using Navigation 3's stable
 * single-pane rendering until a responsive multi-pane scene is introduced.
 */
class SplitViewSceneStrategy(
    val splitRatio: Float = 0.4f,
) : NavigationSceneStrategy<PassVaultRoute> {
    init {
        require(splitRatio in 0f..1f) {
            "splitRatio must be between 0 and 1"
        }
    }

    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? {
        val hasSplitEntry = entries.any {
            it.isListEntry() || it.isDetailEntry()
        }

        return if (hasSplitEntry) {
            singlePaneScene(this, entries)
        } else {
            null
        }
    }
}

/**
 * Default PassVault scene strategy.
 */
class VaultMainSceneStrategy : NavigationSceneStrategy<PassVaultRoute> {
    override fun SceneStrategyScope<PassVaultRoute>.calculateScene(
        entries: List<NavEntry<PassVaultRoute>>,
    ): Scene<PassVaultRoute>? = singlePaneScene(this, entries)
}

fun NavEntry<*>.isListEntry(): Boolean =
    metadata["is_list"] == true

fun NavEntry<*>.isDetailEntry(): Boolean =
    metadata["is_detail"] == true

fun NavEntry<*>.isDialogEntry(): Boolean =
    metadata["is_dialog"] == true

fun NavEntry<*>.isBottomSheetEntry(): Boolean =
    metadata["is_bottom_sheet"] == true

operator fun NavigationSceneStrategy<PassVaultRoute>.plus(
    other: NavigationSceneStrategy<PassVaultRoute>,
): NavigationSceneStrategy<PassVaultRoute> {
    return SceneStrategies.chain(this, other)
}
